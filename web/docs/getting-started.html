🚀 Getting Started with Ferric

Installation

Prerequisites

  • Rust 1.75+ (rustup update)
  • wasm-pack (cargo install wasm-pack)
  • Node.js 18+ (for development server)

Install Ferric CLI

cargo install ferric-cli

Create a New Project

ferric new my-app
cd my-app

Run Development Server

ferric serve

Your app will be available at http://localhost:8080

Your First App

Create a Component

use ferric::prelude::*;

#[component(selector = "app-root")]
struct AppComponent {
    count: Signal<i32>,
}

impl Injectable for AppComponent {
    fn create(_injector: &Injector) -> Self {
        Self {
            count: signal(0),
        }
    }
}

impl AppComponent {
    fn increment(&self) {
        self.count.update(|n| *n += 1);
    }
}

impl Component for AppComponent {
    fn template(&self) -> String {
        html! {
            <div>
                <h1>Count: {self.count.get()}</h1>
                <button (click)="increment()">Increment</button>
            </div>
        }
    }
}

Bootstrap the App

use wasm_bindgen::prelude::*;

#[wasm_bindgen(start)]
pub fn main() {
    let injector = Injector::root();
    let app = injector.resolve::<AppComponent>().unwrap();
    app.mount("#app");
}

Project Structure

my-app/
├── Cargo.toml
├── src/
│   ├── lib.rs           # Entry point
│   ├── app/
│   │   ├── mod.rs       # App module
│   │   └── component.rs # App component
│   ├── components/      # Reusable components
│   ├── services/        # Injectable services
│   └── models/          # Data models
├── public/
│   ├── index.html
│   └── assets/
└── tests/
